```rust
/// §8.2.4.2.3: apply `modification_of_pic_nums_idc` operations in order.
///
/// `list` holds DPB slots in initial-list order and is modified in place.
/// `curr_frame_num` is the slice's `frame_num`, `max_frame_num` is
/// `1 << log2_max_frame_num`.
pub fn reorder(
    list: &mut Vec<u8>,
    mods: &[RefMod],
    curr_frame_num: u32,
    max_frame_num: u32,
    frames: &[FrameStore],
) {
    let mut insert_at = 0usize;
    for m in mods {
        match *m {
            RefMod::End => break,
            RefMod::Sub(d) => {
                let pic_num = curr_frame_num.wrapping_sub(d + 1) % max_frame_num;
                if let Some(pos) = list.iter().position(|&slot| {
                    frames
                        .iter()
                        .find(|f| f.slot == slot)
                        .map(|f| !f.long_term && f.frame_num % max_frame_num == pic_num)
                        .unwrap_or(false)
                }) {
                    let v = list.remove(pos);
                    list.insert(insert_at.min(list.len()), v);
                    insert_at += 1;
                }
            }
            RefMod::Add(d) => {
                let pic_num = (curr_frame_num + d + 1) % max_frame_num;
                if let Some(pos) = list.iter().position(|&slot| {
                    frames
                        .iter()
                        .find(|f| f.slot == slot)
                        .map(|f| !f.long_term && f.frame_num % max_frame_num == pic_num)
                        .unwrap_or(false)
                }) {
                    let v = list.remove(pos);
                    list.insert(insert_at.min(list.len()), v);
                    insert_at += 1;
                }
            }
            RefMod::LongTerm(pic_num) => {
                if let Some(pos) = list.iter().position(|&slot| {
                    frames
                        .iter()
                        .find(|f| f.slot == slot)
                        .map(|f| f.long_term && f.long_term_frame_idx == pic_num)
                        .unwrap_or(false)
                }) {
                    let v = list.remove(pos);
                    list.insert(insert_at.min(list.len()), v);
                    insert_at += 1;
                }
            }
        }
    }
}
```

#!/usr/bin/env bash
# H.264 regression: build on board, decode every stream, compare with reference.
set -euo pipefail
streams=(
  "/tmp/test720.264:/tmp/o720_ref.yuv"
  "/tmp/test1080.264:/tmp/o1080_ref.yuv"
  "/tmp/test_hard.264:/tmp/o_hard_ref.yuv"

fail=0
for pair in "${streams[@]}"; do
  in="${pair%%:*}"; ref="${pair##*:}"
  out="${ref/ref/rust}"
  ssh t6 "cd ~/rk-mpp-rs && sudo ./target/release/rk-mpp-rs h264dec $in $out" >/dev/null 2>&1 || true

  else
    printf 'FAIL %s\n' "$in"

  fi
done
exit $fail
